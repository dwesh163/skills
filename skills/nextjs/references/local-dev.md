# Local dev stack: Makefile and Docker Compose

## Contents

- Secrets: a shared `secrets.yml`, read by the Makefile
- `Makefile`
- `docker-compose.dev.yml`

The concrete instance, for this stack (Next.js dev server on Node, Bun as package manager,
Postgres), of the rules in the `compose` skill: the whole stack in Docker Compose,
`make up` as the only command, `.env` generated by `make env` from the team's `secrets.yml`.
Read that skill for the why; copy these files for the how.

Commit format, tags and version-bump commits are in the `git` skill.

## Secrets: a shared `secrets.yml`, read by the Makefile

Secrets are not in the repo and not copied around by hand. When the team keeps them in Keybase, the
Makefile reads one `secrets.yml` from the team folder (`/keybase/team/<team>/`), a top-level key per
environment, the same shape in each:

```yaml
# /keybase/team/<team>/secrets.yml
dev:
  auth:
    id: <oidc client id>
    secret: <oidc client secret>
    issuer: https://login.microsoftonline.com/<tenant-id>/v2.0
  database:
    username: app
    password: <password>
    name: app_dev
prod:
  # ...same keys
```

## `Makefile`

```makefile
SHELL := /bin/bash

# Usage:
#   make help                    Show this help
#   make env                     Generate .env from the secrets file, fails if a key is missing
#   make up                      Generate .env, build and run the whole dev stack (Postgres + Next.js) in Docker
#   make down                    Stop the dev stack
#   make logs                    Follow the Next.js logs
#   make all                     Alias for `make up`
#   make clean                   Stop containers, remove volumes and the .next cache
#   make ENV=staging env         Read the `staging:` section of the secrets file instead of `dev:`
#   make YQ_FLAVOR=go env        Force the yq flavour (go|python) instead of auto-detecting it
#   make DEBUG=1 up              Run a target with `set -x` tracing

.DEFAULT_GOAL := help
.PHONY: help env up down logs all clean

ENV            ?= dev
# One file for every environment, a top-level key each (`dev:`, `staging:`).
SECRETS_FILE   ?= /keybase/team/<team>/secrets.yml
ENV_FILE       ?= .env
COMPOSE_FILE   ?= docker-compose.dev.yml
DOCKER_COMPOSE ?= docker compose
COMPOSE        = $(DOCKER_COMPOSE) -f $(COMPOSE_FILE)

# Ports and URLs of the local stack. Non-default Postgres port, so it never collides with a
# Postgres already running natively on the developer's machine.
POSTGRES_PORT ?= 58432
APP_PORT      ?= 3000
AUTH_URL      ?= http://localhost:$(APP_PORT)

export POSTGRES_PORT APP_PORT

# ENV_VAR=path pairs, read from the ENV section of SECRETS_FILE. One line per variable the app
# reads through required() that is a secret (not a local value like AUTH_URL).
SECRETS = \
	CLIENT_ID=auth.id \
	CLIENT_SECRET=auth.secret \
	IDENTITY_ISSUER=auth.issuer \
	POSTGRES_USER=database.username \
	POSTGRES_PASSWORD=database.password \
	POSTGRES_DB=database.name

# Variables defined above, written to ENV_FILE as is.
LOCAL_VARS = AUTH_URL POSTGRES_PORT

# yq comes in two incompatible flavours: mikefarah/yq (Go) and kislyuk/yq (Python, a jq wrapper).
# Both are supported: the flavour is detected from `yq --version` unless YQ_FLAVOR is given.
YQ_FLAVOR := $(or $(YQ_FLAVOR),$(if $(findstring mikefarah,$(shell yq --version 2>&1)),go,python))
$(if $(filter go python,$(YQ_FLAVOR)),,$(error YQ_FLAVOR must be "go" or "python", got "$(YQ_FLAVOR)"))
YQ := $(if $(filter go,$(YQ_FLAVOR)),yq eval,yq -r)

.SHELLFLAGS := -e$(if $(filter 1,$(DEBUG)),x)c

## Print this help (see <https://gist.github.com/klmr/575726c7e05d8780505a>)
help:
	@echo "$$(tput bold)Available rules (alphabetical order):$$(tput sgr0)";sed -ne"/^## /{h;s/.*//;:d" -e"H;n;s/^## //;td" -e"s/:.*//;G;s/\\n## /---/;s/\\n/ /g;p;}" ${MAKEFILE_LIST}|LC_ALL='C' sort -f |awk -F --- -v n=$$(tput cols) -v i=20 -v a="$$(tput setaf 6)" -v z="$$(tput sgr0)" '{printf"%s%*s%s ",a,-i,$$1,z;m=split($$2,w," ");l=n-i;for(j=1;j<=m;j++){l-=length(w[j])+1;if(l<= 0){l=n-i-length(w[j])-1;printf"\n%*s ",-i," ";}printf"%s ",w[j];}printf"\n";}'

## Generate .env from the ENV section of SECRETS_FILE, fails if a key is missing
env:
	@test -f "$(SECRETS_FILE)" || { echo "Missing $(SECRETS_FILE) - sync it from Keybase first" >&2; exit 1; }
	@command -v yq >/dev/null || { echo "Missing yq - install either mikefarah/yq (Go) or kislyuk/yq (Python)" >&2; exit 1; }
	@tmp="$(ENV_FILE).tmp"; trap 'rm -f "$$tmp"' EXIT; \
	{ \
		echo "# Generated by 'make env' from $(SECRETS_FILE)"; \
		for pair in $(SECRETS); do \
			var=$${pair%%=*}; key=.$(ENV).$${pair#*=}; \
			val=$$($(YQ) "$$key" < "$(SECRETS_FILE)") || val=null; \
			[ "$$val" != null ] || { echo "Missing key $$key ($$var) in $(SECRETS_FILE)" >&2; exit 1; }; \
			printf -v "$$var" '%s' "$$val"; \
			echo "$$var=$$val"; \
		done; \
		echo "DATABASE_URL=postgresql://$$POSTGRES_USER:$$POSTGRES_PASSWORD@localhost:$(POSTGRES_PORT)/$$POSTGRES_DB?schema=public"; \
		echo; \
		: "AUTH_SECRET is not shared: keep the one already in ENV_FILE (sessions survive), or generate one"; \
		secret=$$(sed -n 's/^AUTH_SECRET=//p' "$(ENV_FILE)" 2>/dev/null || true); \
		echo "AUTH_SECRET=$${secret:-$$(openssl rand -base64 32)}"; \
		$(foreach v,$(LOCAL_VARS),echo "$(v)=$($(v))";) \
	} > "$$tmp"; \
	mv "$$tmp" "$(ENV_FILE)"; \
	echo "Wrote $(ENV_FILE) from $(SECRETS_FILE)"

## Generate .env and run the whole dev stack (Postgres + Next.js) in Docker
up: env
	$(COMPOSE) up --build

## Stop the dev stack
down:
	$(COMPOSE) down

## Follow the Next.js logs of the dev stack
logs:
	$(COMPOSE) logs -f app

## Generate .env and start the full dev stack
all: up

## Stop containers, remove volumes and the .next cache
clean:
	$(COMPOSE) down -v
	rm -rf .next
```

Why each part of the recipe is there (fail on a missing key, atomic write, per-developer
`AUTH_SECRET`, derived `DATABASE_URL`, both `yq` flavours): see the `compose` skill. One
detail specific to this recipe: the `|| true` after the `sed` that reads the existing
`AUTH_SECRET` keeps the very first `make env` on a fresh clone from dying under `-e`.

**No shared secret store** (nothing to put in Keybase, or a Postgres-only app)? Drop `SECRETS_FILE`,
`SECRETS` and `YQ`: `env` copies `.env.example` to `.env` only when `.env` is missing, with
`AUTH_SECRET` filled by `openssl rand -base64 32`. Everything else stays identical.

## `docker-compose.dev.yml`

```yaml
services:
  postgres:
    image: postgres:18-alpine
    restart: unless-stopped
    ports:
      - "${POSTGRES_PORT:-58432}:5432"
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-postgres}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-postgres}
      POSTGRES_DB: ${POSTGRES_DB:-app_dev}
    volumes:
      # Since Postgres 18 the image keeps its cluster in /var/lib/postgresql/18/docker, not in
      # /var/lib/postgresql/data: mounting the latter leaves the data in an anonymous volume that
      # is lost whenever the container is recreated.
      - postgres_data:/var/lib/postgresql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-postgres} -d ${POSTGRES_DB:-app_dev}"]
      interval: 2s
      timeout: 3s
      retries: 15

  # Named volumes are created root-owned: hand them over to the `node` user (uid 1000, the usual
  # host uid) that the app runs as, so files written to the bind mount stay owned by the host user.
  init:
    image: node:26-alpine
    user: root
    command: chown node:node /app/node_modules /app/.next
    volumes:
      - node_modules:/app/node_modules
      - next_cache:/app/.next

  app:
    image: node:26-alpine
    # The first step downloads Bun: a network blip at that moment (EAI_AGAIN) would otherwise leave
    # the app down until the next `make up`.
    restart: on-failure:3
    user: node
    working_dir: /app
    ports:
      - "${APP_PORT:-3000}:3000"
    env_file: .env
    environment:
      # .env points at localhost:${POSTGRES_PORT} (for tools on the host); inside the compose
      # network Postgres is reached through its service name.
      DATABASE_URL: postgresql://${POSTGRES_USER:-postgres}:${POSTGRES_PASSWORD:-postgres}@postgres:5432/${POSTGRES_DB:-app_dev}?schema=public
      HOSTNAME: 0.0.0.0
      NEXT_TELEMETRY_DISABLED: "1"
      # Node is what Next.js runs on, Bun is the package manager and script runner as on the host:
      # install it in the `node` user's home.
      NPM_CONFIG_PREFIX: /home/node/.npm-global
      NPM_CONFIG_UPDATE_NOTIFIER: "false"
    # --frozen-lockfile keeps bun.lock untouched (and fails loudly if it is out of sync).
    # `bun run dev` already runs `prisma generate`; drop the migrate line without Prisma.
    command:
      - sh
      - -ec
      - |
        export PATH="$$NPM_CONFIG_PREFIX/bin:$$PATH"
        npm install --global bun
        bun install --frozen-lockfile
        bunx prisma migrate deploy
        exec bun run dev
    depends_on:
      init:
        condition: service_completed_successfully
      postgres:
        condition: service_healthy
    volumes:
      - .:/app
      # Native deps are built for musl in the container: keep them out of the host's node_modules.
      - node_modules:/app/node_modules
      - next_cache:/app/.next

volumes:
  postgres_data:
  node_modules:
  next_cache:
```

Specific to this stack (the generic pitfalls are in the `compose` skill):

- **Keep `AUTH_URL` and `APP_PORT` in sync.** better-auth rejects callbacks from any origin other
  than `AUTH_URL`, and `AUTH_URL` + `/api/auth/callback/<providerId>` is the redirect URI to
  register on the IdP app (auth.md).
- **Since Postgres 18** the image keeps its cluster under `/var/lib/postgresql/<major>/docker`:
  mount the volume at `/var/lib/postgresql`, not `/var/lib/postgresql/data`.

`package.json`'s `dev` script stays plain (`bunx prisma generate && next dev`): compose runs it, the
Makefile starts compose.
